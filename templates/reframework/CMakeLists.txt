cmake_minimum_required(VERSION 3.24)
project(BoneforgeREMenu VERSION 0.1.0 LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use an x64 MSVC toolchain for the REFramework plug-in")
endif()
set(REF_INCLUDE_DIR "" CACHE PATH "Directory containing reframework/API.h")
set(REF_LUA_DIR "" CACHE PATH "Matching REFramework dependencies/lua/src directory")
if(NOT EXISTS "${REF_INCLUDE_DIR}/reframework/API.h" OR NOT EXISTS "${REF_LUA_DIR}/lapi.c")
    message(FATAL_ERROR "Supply REF_INCLUDE_DIR and REF_LUA_DIR from the same supported REFramework source revision")
endif()
set(CMAKE_CXX_STANDARD 17)
file(GLOB luaSources CONFIGURE_DEPENDS "${REF_LUA_DIR}/*.c")
list(FILTER luaSources EXCLUDE REGEX "/(lua|luac|onelua)\\.c$")
add_library(ref_lua STATIC ${luaSources})
target_include_directories(ref_lua PUBLIC "${REF_LUA_DIR}")

configure_file(identity.h.in identity.h @ONLY)
set(autorun "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.lua")
configure_file(Menu.lua.in "${autorun}" @ONLY)
add_library(${PROJECT_NAME} SHARED Plugin.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE "${REF_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(${PROJECT_NAME} PRIVATE ref_lua)
install(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION reframework/plugins)
install(FILES "${autorun}" DESTINATION reframework/autorun)
install(FILES LICENSE LICENSE.reframework.txt LICENSE.lua.txt DESTINATION "licenses/${PROJECT_NAME}")

include(CTest)
if(BUILD_TESTING)
    add_executable(host_test HostTest.cpp)
    target_include_directories(host_test PRIVATE "${REF_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(host_test PRIVATE ref_lua)
    add_test(NAME lifecycle COMMAND host_test "$<TARGET_FILE:${PROJECT_NAME}>" "${autorun}"
        "${CMAKE_CURRENT_SOURCE_DIR}/test_ui.lua")
endif()
